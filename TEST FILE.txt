import { Component, inject, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule, FormBuilder, Validators } from '@angular/forms';
import { Router } from '@angular/router';
import { HealthLogService } from '../services/health-log.service';
import { Gender } from '../models/health-log.model';

function toNumberOrNull(v: unknown): number | null {
  if (v === null || v === undefined || v === '') return null;
  const n = Number(v);
  return isNaN(n) ? null : n;
}

const genderPattern = /^(male|female|other)$/;

@Component({
  selector: 'app-health-log-form',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: './health-log-form.html',
  styleUrls: ['./health-log-form.css'],
})
export class HealthLogForm {
  private fb = inject(FormBuilder);
  private store = inject(HealthLogService);
  private router = inject(Router);

  form = this.fb.nonNullable.group({
    name: ['', [Validators.required, Validators.minLength(2), Validators.maxLength(60)]],
    age: [null, [Validators.required, Validators.min(0), Validators.max(120)]],

    gender: ['', [Validators.required, Validators.pattern(genderPattern)]],

    heightFt: [null, [Validators.required, Validators.min(1), Validators.max(8)]],
    heightIn: [0, [Validators.required, Validators.min(0), Validators.max(11)]],

    weightKg: [null, [Validators.required, Validators.min(2), Validators.max(400)]],

    healthConcern: ['', [Validators.required, Validators.maxLength(160)]],
  });

  statusText = signal<string | null>(null);

  private toCentimeters(ft: number | null, inch: number | null): number | null {
    if (ft == null || ft <= 0) return null;
    const inchesTotal = (ft * 12) + (inch ?? 0);
    if (inchesTotal <= 0) return null;
    return +(inchesTotal * 2.54).toFixed(1);
  }

  onFieldInput() {
    const ft = toNumberOrNull(this.form.controls.heightFt.value);
    const inc = toNumberOrNull(this.form.controls.heightIn.value) ?? 0;
    const cm = this.toCentimeters(ft, inc);
    const w = toNumberOrNull(this.form.controls.weightKg.value);

    if (cm && w && this.form.controls.weightKg.valid) {
      this.statusText.set(this.store.statusText(cm, w));
    } else {
      this.statusText.set(null);
    }
  }

  touchedInvalid(ctrl: keyof typeof this.form.controls) {
    const c = this.form.controls[ctrl];
    return c.touched && c.invalid;
  }

  /** classify banner color */
  statusKind(): 'healthy' | 'under' | 'over' | null {
    const t = this.statusText();
    if (!t) return null;
    if (t.startsWith('Healthy')) return 'healthy';
    if (t.startsWith('Underweight')) return 'under';
    if (t.startsWith('Overweight')) return 'over';
    return null;
  }

  onSubmit() {
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      return;
    }

    const { name, age, gender, heightFt, heightIn, weightKg, healthConcern } =
      this.form.getRawValue();

    const feet = toNumberOrNull(heightFt)!;
    const inches = toNumberOrNull(heightIn) ?? 0;
    const cm = this.toCentimeters(feet, inches);

    if (!cm) {
      this.form.controls.heightFt.setErrors({ invalidHeight: true });
      return;
    }

    this.store.add({
      name: name.trim(),
      age: +age!,
      gender: gender as Gender,
      heightCm: cm,
      weightKg: +weightKg!,
      healthConcern: healthConcern!.trim(),
    });

    this.router.navigate(['/dashboard']);
  }
}

    this._logs.set([]);
  }

  /** Healthy weight range from height using BMI 18.5–24.9. */
  healthyRangeKg(heightCm: number): { min: number; max: number } {
    const m = heightCm / 100;
    const min = +(18.5 * m * m).toFixed(1);
    const max = +(24.9 * m * m).toFixed(1);
    return { min, max };
  }

  /** Returns a friendly text like “Overweight — lose 5.2 kg (range 58.3–78.4 kg)”. */
  statusText(heightCm: number, weightKg: number): string | null {
    if (!heightCm || !weightKg) return null;
    const { min, max } = this.healthyRangeKg(heightCm);
    if (weightKg < min) {
      const gain = +(min - weightKg).toFixed(1);
      return `Underweight - gain ${gain} kg (healthy range ${min}–${max} kg)`;
    }
    if (weightKg > max) {
      const lose = +(weightKg - max).toFixed(1);
      return `Overweight - lose ${lose} kg (healthy range ${max}–${min} kg)`.replace(
        `${max}–${min}`, `${min}–${max}`
      );
    }
    return `Healthy weight range (${min}–${max} kg)`;
  }
}
